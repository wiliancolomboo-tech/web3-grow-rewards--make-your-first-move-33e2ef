FROM node:20-alpine AS builder

WORKDIR /usr/src/app
COPY relayer-service/package*.json ./
RUN npm ci

COPY relayer-service/ ./
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production

COPY relayer-service/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist
COPY --from=builder /usr/src/app/src/idl ./dist/idl

USER node
EXPOSE 8080

CMD ["node", "dist/daemon.js"]
