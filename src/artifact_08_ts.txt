import express from "express";
import { z } from "zod";

const app = express();
app.use(express.json({ limit: "256kb" }));

const emailSchema = z.object({
  wallet: z.string().min(32).max(64),
  email: z.string().email()
});

app.get("/healthz", (_req, res) => {
  res.status(200).json({
    ok: true,
    service: "grow-rewards-ops",
    timestamp: new Date().toISOString()
  });
});

app.get("/readyz", (_req, res) => {
  const configured = Boolean(
    process.env.SOLANA_RPC_URL &&
    process.env.CAMPAIGN_ADDRESS
  );

  res.status(configured ? 200 : 503).json({ ready: configured });
});

app.post("/v1/notification-recipient", (req, res) => {
  const parsed = emailSchema.safeParse(req.body);

  if (!parsed.success) {
    return res.status(400).json({
      error: "Invalid request",
      details: parsed.error.flatten()
    });
  }

  // In production, persist this mapping in PostgreSQL.
  // The wallet remains the canonical identity; email is optional metadata.
  return res.status(202).json({
    accepted: true,
    wallet: parsed.data.wallet
  });
});

const port = Number(process.env.PORT ?? 8080);
app.listen(port, "0.0.0.0", () => {
  console.log(`Grow Rewards service listening on ${port}`);
});
