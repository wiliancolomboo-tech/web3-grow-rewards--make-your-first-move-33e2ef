# Multi-stage production build for GrowRewards Event Engine
FROM node:20-alpine AS builder

WORKDIR /usr/src/app
COPY package*.json ./
COPY tsconfig.json ./
RUN npm ci

COPY . .
RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production
COPY package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist
COPY --from=builder /usr/src/app/target ./target

EXPOSE 8080
USER node

CMD ["node", "dist/engine-daemon.js"]
